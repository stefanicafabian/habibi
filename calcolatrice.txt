void operazioni_base() {
    cout << "\n1. addizione (+)\n";
    cout << "2. sottrazione (-)\n";
    cout << "3. moltiplicAzione (*)\n";
    cout << "4. divisione (/)\n";

    int scelta;
    cout << "\nscegli operazione: ";

    if (!(cin >> scelta)) {
        cin.clear();
        cin.ignore(numeric_limits<streamsize>::max(), '\n');
        cout << "input sussy 💀\n";
        return;
    }

    if (scelta < 1 || scelta > 4) {
        cout << "bro ha scelto il nulla 💀\n";
        return;
    }

    double a = leggi_numero("primo numero: ");
    double b = leggi_numero("secondo numero: ");
    double risultato = 0;
    string operazione;

    switch (scelta) {
        case 1:
            risultato = a + b;
            operazione = "+";
            break;
        case 2:
            risultato = a - b;
            operazione = "-";
            break;
        case 3:
            risultato = a * b;
            operazione = "*";
            break;
        case 4:
            if (b == 0) {
                cout << "\nbro ha diviso per zero 💀\n";
                return;
            }
            risultato = a / b;
            operazione = "/";
            break;
    }

    if (!isfinite(risultato)) {
        cout << "risultato fuori dai limiti Numerici.\n";
        return;
    }

    mostra_risultato(risultato,
        to_string(a) + " " + operazione + " " + to_string(b));
}


void operazioni_avanzate() {

    cout << "11. pi greco\n";

    int scelta;
    cout << "\nscegli: ";

    if (!(cin >> scelta)) {
        cin.clear();
        cin.ignore(numeric_limits<streamsize>::max(), '\n');
        cout << "bro, inpuT in crisi 😭\n";
        return;
    }

    double a, b, risultato = 0;
    string operazione;

    switch (scelta) {
        case 1:
            a = leggi_numero("base: ");
            b = leggi_numero("esponente: ");
            risultato = pow(a, b);
            operazione = "potenza";
            break;

        case 2:
            a = leggi_numero("numero: ");
            if (a < 0) {
                cout << "radice reale impossibile.\n";
                return;
            }
            risultato = sqrt(a);
            operazione = "sqrt";
            break;

        case 3: {
            long long x, y;

            cout << "primo intero: ";
            if (!(cin >> x)) {
                cin.clear();
                cin.ignore(numeric_limits<streamsize>::max(), '\n');
                cout << "intero non valido.\n";
                return;
            }

            cout << "secondo intero: ";
            if (!(cin >> y)) {
                cin.clear();
                cin.ignore(numeric_limits<streamsize>::max(), '\n');
                cout << "intero non valido.\n";
                return;
            }

            if (y == 0) {
                cout << "modulo per zero? bro no 💀\n";
                return;
            }

            if (x == numeric_limits<long long>::min() && y == -1) {
                cout << "risultato fuori intervallo.\n";
                return;
            }

            risultato = static_cast<double>(x % y);
            operazione = "modulo";
            break;
        }

        case 4:
            a = leggi_numero("percentuale: ");
            b = leggi_numero("numero di riferimento: ");
            risultato = a * b / 100.0;
            operazione = "percentuale";
            break;

        case 5:
        case 6:
        case 7: {
            a = leggi_numero("angolo in gradi: ");
            double radianti = a * pi / 180.0;

            if (scelta == 5) {
                risultato = sin(radianti);
                operazione = "sin";
            } else if (scelta == 6) {
                risultato = cos(radianti);
                operazione = "cos";
            } else {
                if (fabs(cos(radianti)) < 1e-12) {
                    cout << "tangente non definita.\n";
                    return;
                }
                risultato = tan(radianti);
                operazione = "tan";
            }
            break;
        }

        case 8:
            a = leggi_numero("numero: ");
            if (a <= 0) {
                cout << "ln richiede un numero positivo.\n";
                return;
            }
            risultato = log(a);
            operazione = "ln";
            break;

        case 9:
            a = leggi_numero("numero: ");
            risultato = fabs(a);
            operazione = "abs";
            break;

        case 10:
            a = leggi_numero("numero: ");
            if (a == 0) {
                cout << "l'inverso di zero non esiste 💀\n";
                return;
            }
            risultato = 1.0 / a;
            operazione = "inverso";
            break;

        case 11:
            risultato = pi;
            operazione = "pi greco";
            break;

        default:
            cout << "scelta non valida, skill issue.\n";
            return;
    }

    if (!isfinite(risultato)) {
        cout << "risultato fuori dai limiti numerici.\n";
        return;
    }

    mostra_risultato(risultato, operazione);
}
